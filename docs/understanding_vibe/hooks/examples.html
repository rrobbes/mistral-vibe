🎯 Exemples Complets de Hooks

Découvrez des exemples prêts à l'emploi pour chaque type de hook. Copiez, collez, et adaptez ces exemples pour vos besoins spécifiques.

📁 Exemples par Catégorie

🎯 Comment utiliser ces exemples ?

  1. Créez un dossier pour vos hooks : 
    mkdir -p ~/.vibe/hooks/security ~/.vibe/hooks/audit ~/.vibe/hooks/transform
  2. Créez les fichiers Python/Bash avec le code des exemples
  3. Rendez-les exécutables : 
    chmod +x ~/.vibe/hooks/**/*.py
  4. Configurez vos hooks dans .vibe/hooks.toml
  5. Testez avec Mistral Vibe

ℹ️ Astuce : Vous pouvez aussi placer les hooks dans ./.vibe/hooks/ à la racine de votre projet pour qu'ils ne s'appliquent qu'à ce projet.

🔒 Exemples de Sécurité

🛡️ Bloquer les Commandes Bash Dangereuses

Type : pre_tool
Match : bash
Strict : true
Description : Bloque les commandes bash dangereuses comme rm -rf, chmod -R, etc.

~/.vibe/hooks/security/block_dangerous_bash.py 
#!/usr/bin/env python3
"""
Hook PRE_TOOL pour bloquer les commandes bash dangereuses.

Ce hook intercepte tous les appels à l'outil 'bash' et vérifie
si la commande contient des patterns dangereux.
"""

import json
import sys
import re


def is_dangerous(command: str) -> tuple[bool, str]:
    """Vérifie si une commande est dangereuse."""
    
    # Patterns dangereux avec leur description
    dangerous_patterns = [
        (r'rm\s+-r[f]?\s+', 'rm -rf ou rm -r'),
        (r'chmod\s+-R\s+', 'chmod -R'),
        (r':\s*\(\s*:\s*\|\s*:\s*\)\s*;', 'Fork bomb'),
        (r'dd\s+if=/dev/', 'dd depuis /dev/'),
        (r'cat\s+/etc/shadow', 'Lecture de /etc/shadow'),
        (r'wget\s+.*\|\s*sh', 'Téléchargement et exécution'),
        (r'curl\s+.*\|\s*sh', 'Téléchargement et exécution'),
        (r'\>\s*/etc/', 'Écriture dans /etc/'),
        (r'mv\s+.+\s+/dev/null', 'Déplacement vers /dev/null'),
        (r'find\s+.*\s+-exec\s+', 'find avec -exec'),
    ]
    
    for pattern, description in dangerous_patterns:
        if re.search(pattern, command, re.IGNORECASE):
            return True, f"Commande dangereuse détectée : {description}"
    
    # Commandes spécifiques dangereuses
    dangerous_commands = [
        'rm -rf /',
        'rm -rf *',
        ':(){ :|:& };:',
        'mkfs',
        'fdformat',
    ]
    
    for cmd in dangerous_commands:
        if cmd in command:
            return True, f"Commande interdite : {cmd}"
    
    return False, ""


def main():
    """Point d'entrée du hook."""
    try:
        # Lire l'invocation depuis stdin
        invocation = json.load(sys.stdin)
        
        # Vérifier si c'est un appel bash
        if invocation.get("tool_name") != "bash":
            # Pas un appel bash, autoriser
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Extraire la commande
        command = invocation.get("tool_input", {}).get("command", "")
        if not command:
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Vérifier si la commande est dangereuse
        is_dangerous_cmd, reason = is_dangerous(command)
        
        if is_dangerous_cmd:
            # Bloquer la commande
            print(json.dumps({
                "decision": "deny",
                "reason": reason
            }))
            sys.exit(0)
        
        # Autoriser la commande
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except json.JSONDecodeError:
        # Erreur de parsing JSON
        print(json.dumps({
            "decision": "deny",
            "reason": "Erreur de parsing de l'invocation"
        }))
        sys.exit(1)
    except Exception as e:
        # Autre erreur
        print(json.dumps({
            "decision": "deny",
            "reason": f"Erreur dans le hook : {str(e)}"
        }))
        sys.exit(1)


if __name__ == "__main__":
    main()

🔐 Restreindre l'Accès aux Fichiers

Type : pre_tool
Match : read_file,write_file
Strict : true
Description : Empêche l'accès aux fichiers en dehors du répertoire de travail.

~/.vibe/hooks/security/restrict_file_access.py 
#!/usr/bin/env python3
"""
Hook PRE_TOOL pour restreindre l'accès aux fichiers.

Ce hook vérifie que les opérations de lecture/écriture
ne sortent pas du répertoire de travail courant.
"""

import json
import sys
from pathlib import Path


def is_allowed_path(target_path: str, cwd: str) -> bool:
    """Vérifie si un chemin est autorisé (dans cwd)."""
    try:
        # Résoudre les chemins
        target = Path(target_path).resolve()
        base = Path(cwd).resolve()
        
        # Vérifier que target commence par base
        return str(target).startswith(str(base))
    except Exception:
        return False


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        # Vérifier si c'est un appel à read_file ou write_file
        tool_name = invocation.get("tool_name")
        if tool_name not in ["read_file", "write_file"]:
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Extraire le path
        path = invocation.get("tool_input", {}).get("path", "")
        if not path:
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Extraire le répertoire de travail
        cwd = invocation.get("cwd", "/")
        
        # Vérifier si le path est autorisé
        if not is_allowed_path(path, cwd):
            print(json.dumps({
                "decision": "deny",
                "reason": f"Accès interdit en dehors de {cwd}"
            }))
            sys.exit(0)
        
        # Autoriser
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except Exception as e:
        print(json.dumps({
            "decision": "deny",
            "reason": f"Erreur dans le hook : {str(e)}"
        }))
        sys.exit(1)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Hook PRE_TOOL pour bloquer certains outils complètement.
"""

import json
import sys


# Liste des outils bloqués
BLOCKED_TOOLS = {
    "bash": "Les commandes shell sont désactivées",
    "python": "L'exécution de code Python est désactivée",
    "sh": "Les scripts shell sont désactivés",
    "zsh": "Les scripts zsh sont désactivés",
    "powershell": "PowerShell est désactivé",
}


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        tool_name = invocation.get("tool_name", "")
        
        if tool_name in BLOCKED_TOOLS:
            print(json.dumps({
                "decision": "deny",
                "reason": BLOCKED_TOOLS[tool_name]
            }))
            sys.exit(0)
        
        # Autoriser par défaut
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except Exception as e:
        print(json.dumps({"decision": "allow"}))  # Ne pas bloquer par erreur
        sys.exit(1)


if __name__ == "__main__":
    main()

📊 Exemples d'Audit

📝 Journaliser Tous les Appels d'Outils

Type : post_tool
Match : * (tous les outils)
Strict : false
Description : Journalise tous les appels d'outils dans un fichier structuré.

~/.vibe/hooks/audit/log_tool_calls.py 
#!/usr/bin/env python3
"""
Hook POST_TOOL pour journaliser tous les appels d'outils.

Ce hook écrit un log structuré de chaque appel d'outil
avec son résultat, sa durée, et son statut.
"""

import json
import sys
from datetime import datetime
from pathlib import Path
import os


LOG_DIR = Path("~/vibe_audit").expanduser()
LOG_FILE = LOG_DIR / "tool_calls.jsonl"


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        # Créer le répertoire de log s'il n'existe pas
        LOG_DIR.mkdir(parents=True, exist_ok=True)
        
        # Créer l'entrée de log
        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "session_id": invocation.get("session_id"),
            "tool_name": invocation.get("tool_name"),
            "tool_call_id": invocation.get("tool_call_id"),
            "tool_input": invocation.get("tool_input", {}),
            "tool_status": invocation.get("tool_status"),
            "tool_output": invocation.get("tool_output"),
            "tool_error": invocation.get("tool_error"),
            "duration_ms": invocation.get("duration_ms"),
            "cwd": invocation.get("cwd"),
        }
        
        # Écrire dans le fichier de log (format JSONL)
        with open(LOG_FILE, "a") as f:
            f.write(json.dumps(log_entry) + "\n")
        
        # Toujours autoriser
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except Exception as e:
        # En cas d'erreur, on autorise mais on loggue l'erreur
        print(json.dumps({"decision": "allow"}))
        sys.exit(1)


if __name__ == "__main__":
    main()
.vibe/hooks.toml 
[[hooks]]

💡 Astuce : Le format JSONL (JSON Lines) est parfait pour les logs :

  • Chaque ligne est un JSON valide
  • Facile à parser avec jq, Python, etc.
  • On peut streamer les logs en temps réel
Pour lire les logs : tail -f ~/vibe_audit/tool_calls.jsonl | jq

🎭 Modifier les Arguments avant Exécution

Type : pre_tool
Match : bash
Strict : false
Description : Ajoute automatiquement --color=always aux commandes ls.

~/.vibe/hooks/transform/add_colors_to_ls.py 
#!/usr/bin/env python3
"""
Hook PRE_TOOL pour ajouter --color=always aux commandes ls.
"""

import json
import sys
import re


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        # Vérifier si c'est un appel bash
        if invocation.get("tool_name") != "bash":
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Extraire la commande
        command = invocation.get("tool_input", {}).get("command", "")
        
        # Vérifier si c'est une commande ls
        if not command.strip().startswith("ls"):
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Ajouter --color=always si pas déjà présent
        if "--color" not in command and "-color" not in command:
            # Remplacer ls par ls --color=always
            new_command = re.sub(r'\bls\b', 'ls --color=always', command, count=1)
            
            print(json.dumps({
                "decision": "allow",
                "hook_specific_output": {
                    "tool_input": {"command": new_command}
                }
            }))
            sys.exit(0)
        
        # Sinon, autoriser sans modification
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except Exception as e:
        print(json.dumps({"decision": "allow"}))
        sys.exit(1)


if __name__ == "__main__":
    main()

🔒 Masquer les Informations Sensibles

Type : post_tool
Match : * (tous les outils)
Strict : false
Description : Masque les mots de passe, tokens, et clés API dans les sorties.

~/.vibe/hooks/transform/mask_secrets.py 
#!/usr/bin/env python3
"""
Hook POST_TOOL pour masquer les informations sensibles.
"""

import json
import sys
import re


# Liste des mots-clés sensibles
SENSITIVE_KEYWORDS = [
    r'password',
    r'passwd',
    r'secret',
    r'token',
    r'apikey',
    r'api_key',
    r'auth',
    r'authorization',
    r'private_key',
    r'access_key',
    r'aws_access_key_id',
    r'aws_secret_access_key',
]


def mask_secrets(text: str) -> str:
    """Masque les informations sensibles dans un texte."""
    
    # Pour chaque mot-clé sensible
    for keyword in SENSITIVE_KEYWORDS:
        # Pattern pour matcher : keyword = valeur ou keyword: valeur
        pattern = rf'({keyword})\s*[=:]\s*["\']?([^\s"\'&\n]+)["\']?'
        replacement = r'\1=[MASKED]'
        text = re.sub(pattern, replacement, text, flags=re.IGNORECASE)
    
    return text


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        # Récupérer la sortie
        output_text = invocation.get("tool_output_text", "")
        
        # Masquer les secrets
        masked_output = mask_secrets(output_text)
        
        # Remplacer la sortie
        print(json.dumps({
            "decision": "deny",
            "reason": masked_output,
            "hook_specific_output": {
                "additional_context": "\n[Les informations sensibles ont été masquées]"
            }
        }))
        sys.exit(0)
        
    except Exception as e:
        print(json.dumps({"decision": "allow"}))
        sys.exit(1)


if __name__ == "__main__":
    main()

💬 Notifier Slack après Chaque Tour

Type : post_agent
Strict : false
Description : Envoie une notification à Slack après chaque tour de conversation.

~/.vibe/hooks/integration/slack_notify.py 
#!/usr/bin/env python3
"""
Hook POST_AGENT pour notifier Slack.

Ce hook envoie une notification à Slack après chaque tour
de conversation. Utilise une variable d'environnement pour
le webhook URL.
"""

import json
import sys
import os
import requests


SLACK_WEBHOOK_URL = os.getenv("SLACK_WEBHOOK_URL")


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        # Si pas de webhook configuré, ignorer
        if not SLACK_WEBHOOK_URL:
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Construire le message
        message = {
            "text": f"✅ Mistral Vibe a terminé un tour",
            "blocks": [
                {
                    "type": "section",
                    "text": {
                        "type": "mrkdwn",
                        "text": f"*✅ Mistral Vibe - Tour Terminé*"
                    }
                },
                {
                    "type": "divider"
                },
                {
                    "type": "section",
                    "fields": [
                        {
                            "type": "mrkdwn",
                            "text": f"*Session:*\n{invocation.get('session_id', 'unknown')}"
                        },
                        {
                            "type": "mrkdwn",
                            "text": f"*Répertoire:*\n{invocation.get('cwd', 'unknown')}"
                        }
                    ]
                },
                {
                    "type": "context",
                    "elements": [
                        {
                            "type": "mrkdwn",
                            "text": " Hooks Integration"
                        }
                    ]
                }
            ]
        }
        
        # Envoyer à Slack
        try:
            response = requests.post(
                SLACK_WEBHOOK_URL,
                json=message,
                timeout=5
            )
            response.raise_for_status()
        except Exception as e:
            # Ne pas bloquer l'agent si Slack échoue
            pass
        
        # Toujours autoriser
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except Exception as e:
        print(json.dumps({"decision": "allow"}))
        sys.exit(1)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Hook POST_TOOL pour déclencher un pipeline CI/CD.

Ce hook vérifie si le fichier écrit est un fichier source
et déclenche un pipeline CI/CD si nécessaire.
"""

import json
import sys
import os
import subprocess
from pathlib import Path


# Extensions de fichiers sources
SOURCE_EXTENSIONS = ['.py', '.js', '.ts', '.go', '.rs', '.java', '.cpp', '.c', '.h']

# Commandes CI/CD
CI_CD_COMMANDS = {
    '.py': 'make test',
    '.js': 'npm test',
    '.ts': 'npm test',
    '.go': 'go test ./...',
}


def should_trigger_ci(file_path: str) -> str | None:
    """Vérifie si on doit déclencher un pipeline CI/CD."""
    
    path = Path(file_path)
    ext = path.suffix.lower()
    
    if ext in SOURCE_EXTENSIONS:
        return CI_CD_COMMANDS.get(ext, 'make test')
    
    return None


def main():
    """Point d'entrée du hook."""
    try:
        invocation = json.load(sys.stdin)
        
        # Vérifier si c'est un appel write_file
        if invocation.get("tool_name") != "write_file":
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Vérifier si on doit déclencher le CI/CD
        path = invocation.get("tool_input", {}).get("path", "")
        command = should_trigger_ci(path)
        
        if not command:
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Vérifier si on est dans un dépôt git
        cwd = invocation.get("cwd", ".")
        git_dir = Path(cwd) / ".git"
        
        if not git_dir.exists():
            print(json.dumps({"decision": "allow"}))
            sys.exit(0)
        
        # Exécuter la commande CI/CD en arrière-plan
        try:
            process = subprocess.Popen(
                ["sh", "-c", command],
                cwd=cwd,
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL
            )
            # On ne wait pas, on retourne immédiatement
        except Exception as e:
            # Ne pas bloquer l'agent
            pass
        
        # Toujours autoriser
        print(json.dumps({"decision": "allow"}))
        sys.exit(0)
        
    except Exception as e:
        print(json.dumps({"decision": "allow"}))
        sys.exit(1)


if __name__ == "__main__":
    main()
.vibe/hooks.toml 
[[hooks]]
name = "integration-trigger-ci-cd"
type = "post_tool"
command = "python3 ~/.vibe/hooks/integration/trigger_ci_cd.py"
match = "write_file"
timeout = 5
strict = false
description = "Déclenche un pipeline CI/CD après write_file"

⚠️ Attention :

  • Ce hook déclenche des commandes en arrière-plan
  • Il ne wait pas la fin de l'exécution
  • Les erreurs ne sont pas visibles dans Mistral Vibe
  • Pour un vrai pipeline, utilisez plutôt des GitHub Actions ou GitLab CI
